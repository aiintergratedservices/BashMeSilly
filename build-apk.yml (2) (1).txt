name: CampLoJack Build and Deploy

on:
  push:
    branches: [ "main", "master" ]
  pull_request:
    branches: [ "main", "master" ]

jobs:
  build-frontend:
    name: Build & Validate Frontend
    runs-on: ubuntu-latest
    steps:
      - name: Checkout Code
        uses: actions/checkout@v4

      - name: Set up Node.js
        uses: actions/setup-node@v4
        with:
          node-version: 18
          cache: 'npm'

      - name: Install Dependencies
        run: npm ci

      - name: Run Linter
        run: npm run lint

      - name: Build Web App
        run: npm run build
        env:
          VITE_SUPABASE_URL: ${{ secrets.VITE_SUPABASE_URL }}
          VITE_SUPABASE_ANON_KEY: ${{ secrets.VITE_SUPABASE_ANON_KEY }}

      - name: Upload Web Dist Artifact
        uses: actions/upload-artifact@v4
        with:
          name: web-dist
          path: dist/

  build-apk:
    name: Compile Sideloadable APK
    runs-on: ubuntu-latest
    steps:
      - name: Checkout Code
        uses: actions/checkout@v4

      - name: Set up JDK 17
        uses: actions/setup-java@v4
        with:
          java-version: '17'
          distribution: 'temurin'
          cache: 'gradle'

      - name: Grant Execute Permission for Gradlew
        run: chmod +x twa-app/gradlew

      - name: Build Debug APK
        run: |
          cd twa-app
          ./gradlew assembleDebug

      - name: Upload Sideloadable APK
        uses: actions/upload-artifact@v4
        with:
          name: camplojack-sideload-apk
          path: twa-app/app/build/outputs/apk/debug/app-debug.apk
